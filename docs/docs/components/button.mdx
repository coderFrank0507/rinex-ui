# Button 按钮
---

### 代码演示

#### 基础
```tsx preview
import { Button } from 'rinex-ui';

export default function App() {
  return (
    <div className="flex flex-wrap gap-2">
      <Button variant="primary">Primary</Button> 
      <Button>Default</Button>
      <Button variant="dashed">Dashed</Button>
      <Button variant="text">Text</Button>
      <Button variant="link">Link</Button>
    </div>
  );
};
```

#### 图标按钮

```tsx preview
import { Button } from 'rinex-ui';

  return (
    <div className="flex flex-wrap gap-2">
      <Button size="icon" className="rounded-full">

      </Button>
      <Button size="icon" variant="primary">
        <Github size={18} />
      </Button>
      <Button>
        <Github size={18} />
        Github
      </Button>
      <Button variant="primary" className="rounded-full">
        <Github size={18} />
        Github
      </Button>
    </div>
  );
};
```

#### 加载状态
```tsx preview
import { useState, useEffect } from 'react';
import { Button } from 'rinex-ui';
import { LoaderCircle, Github } from 'lucide-react';

export default function App() {
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    if (loading) {
      setTimeout(() => {
        setLoading(false);
      }, 2000);
    }
  }, [loading]);
  
  return (
    <div>
      <div className="border-b pb-2 mb-2">
        <Button variant="primary" className="w-full" onClick={() => setLoading(true)}>
          Loading
        </Button>
      </div>
      <div className="flex flex-wrap gap-2">
        <Button disabled={loading} size="icon">
          {loading ?
            <LoaderCircle className="animate-spin" size={18} /> :
            <Github size={18} />}
        </Button>
        <Button disabled={loading}>
          {loading && <LoaderCircle className="animate-spin" size={18} />}
          Loading
        </Button>
      </div>
    </div>
  );
};
```

### Props
| 属性 | 说明 | 类型 | 默认值 |
|--|--|:--:|:--:|
|variant|按钮类型|`primary` `default` `dashed` <br/> `text` `link`|default|
|danger|是否为危险按钮|boolean|false|
|size|按钮大小|`lg` `default` `sm` <br/> `icon` `icon-lg` `icon-sm`|default|
